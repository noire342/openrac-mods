cmake_minimum_required(VERSION 3.20)
project(clank_unleashed LANGUAGES C CXX)
if(NOT OpenRACMod_DIR)
  if(NOT OPENRAC_SDK)
    set(OPENRAC_SDK "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/openrac-mod-sdk-1.10.1")
  endif()
  set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
endif()
find_package(OpenRACMod 1.10 REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/compatibility.json" compatibility)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/openrac-mod.json" manifest)
string(JSON profile_version GET "${manifest}" version)
foreach(field region payload_sha256 disc_bytes register_class client_sha256 on_killed vehicle_update exit_full terrain_update)
  string(JSON profile_${field} GET "${compatibility}" ${field})
endforeach()
foreach(field show remove game_time_step air_offset scale_ticks icon_sha256 icon_lookup warning rendered_frame_count frame_scale_sq allocate_voice voice_pool align slide frame sprite rect width height alarm_bank_load alarm_bank_unload alarm_voice_kill alarm_bank_sha256 alarm_definition_sha256)
  string(JSON profile_hud_${field} GET "${compatibility}" hud ${field})
endforeach()
foreach(field hero resident_class_count class_to_slot moby_pool class_resources)
  string(JSON profile_${field} GET "${compatibility}" globals ${field})
endforeach()
foreach(field state motion transitions hit capsule)
  string(JSON profile_${field} GET "${compatibility}" full_functions ${field})
endforeach()
string(JSON companion_count LENGTH "${compatibility}" native_companions)
math(EXPR companion_last "${companion_count}-1")
set(profile_companions "")
foreach(index RANGE ${companion_last})
  string(JSON companion_id GET "${compatibility}" native_companions ${index} id)
  string(JSON companion_sequences GET "${compatibility}" native_companions ${index} sequences)
  string(APPEND profile_companions "{${companion_id},${companion_sequences}},")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/compatibility.json" "${CMAKE_CURRENT_SOURCE_DIR}/README.md" "${CMAKE_CURRENT_SOURCE_DIR}/NOTICE.md")
configure_file(src/game_profile.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/game_profile.hpp" @ONLY)
include(CTest)
option(CLANK_BUILD_QA "Build isolated PAL telemetry/fixtures (never shipped)" OFF)
if(CLANK_BUILD_QA)
  openrac_add_mod(clank_pal_qa MANIFEST tests/pal-probe-mod.json SOURCES tests/pal_probe.cpp)
  target_compile_features(clank_pal_qa PRIVATE cxx_std_17)
endif()
add_library(clank_policy STATIC src/controller.cpp)
target_include_directories(clank_policy PUBLIC src)
target_include_directories(clank_policy PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_features(clank_policy PUBLIC cxx_std_17)
if(BUILD_TESTING)

if(WIN32)
  if(NOT OPENRAC_MOD_PLATFORM STREQUAL "windows-x86_64")
    message(FATAL_ERROR "Clank libero currently supports Windows x64 only")
  endif()
  set(utility "${CMAKE_CURRENT_SOURCE_DIR}/third_party/file-overlay")
  set(json_root "${CMAKE_CURRENT_SOURCE_DIR}/third_party/openrac")
  if(NOT EXISTS "${utility}/src/windows_overlay.cpp")
    set(utility "${CMAKE_CURRENT_SOURCE_DIR}/../arsenal-file-overlay")
  endif()
  if(NOT EXISTS "${json_root}/common/json.cpp")
    set(json_root "${CMAKE_CURRENT_SOURCE_DIR}/../revolverator/third_party/openrac")
  endif()
  add_library(clank_json STATIC ${json_root}/common/json.cpp)
  target_compile_features(clank_json PRIVATE cxx_std_20)
  target_include_directories(clank_json PUBLIC ${json_root})
  add_library(clank_minhook STATIC ${utility}/third_party/minhook/src/buffer.c ${utility}/third_party/minhook/src/hook.c
    ${utility}/third_party/minhook/src/trampoline.c ${utility}/third_party/minhook/src/hde/hde64.c)
  target_include_directories(clank_minhook PUBLIC ${utility}/third_party/minhook/include)
  openrac_add_mod(clank_unleashed MANIFEST openrac-mod.json SOURCES
    src/plugin.cpp src/native_world.cpp src/native_control.cpp src/native_hud.cpp src/hud_assets.cpp src/alarm_sound.cpp src/resources.cpp src/disc_reader.cpp src/wad.cpp src/asset_hash.cpp
    ${utility}/src/overlay_core.cpp ${utility}/src/windows_overlay.cpp)
  target_include_directories(clank_unleashed PRIVATE src ${utility}/src ${utility}/include ${json_root})
  target_compile_definitions(clank_unleashed PRIVATE FILE_OVERLAY_SUPPORTED_HASH="${profile_client_sha256}")
  target_link_libraries(clank_unleashed PRIVATE clank_policy clank_minhook clank_json bcrypt shell32)
  target_compile_options(clank_unleashed PRIVATE /W4 /EHsc)
  get_target_property(package_dir clank_unleashed OPENRAC_MOD_PACKAGE_DIR)
  configure_file(NOTICE.md "${package_dir}/COPYING-ThirdParty.md" COPYONLY)
  configure_file("${utility}/third_party/minhook/LICENSE.txt" "${package_dir}/LICENSE-MinHook.txt" COPYONLY)
  configure_file("${utility}/docs/LICENSE-MIT.md" "${package_dir}/LICENSE-FileOverlay-API-MIT.txt" COPYONLY)
endif()
