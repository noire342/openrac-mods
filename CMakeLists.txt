cmake_minimum_required(VERSION 3.20)
project(NoireOpenRACMods LANGUAGES C CXX)

if(NOT OpenRACMod_DIR)
  if(DEFINED OPENRAC_SDK)
    set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
  else()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk.lock" lock)
    string(JSON sdk_version GET "${lock}" version)
    set(OpenRACMod_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/openrac-mod-sdk-${sdk_version}/cmake")
  endif()
endif()
find_package(OpenRACMod 1.3 REQUIRED)
include(CTest)

option(BUILD_WEAPON_XP "Build Weapon Experience" ON)
option(BUILD_REVOLVERATOR "Build additive Revolverator development source" OFF)
if(BUILD_WEAPON_XP)
  add_subdirectory(weapon-xp)
endif()
if(BUILD_REVOLVERATOR)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/revolverator/CMakeLists.txt")
    message(FATAL_ERROR "The Revolverator development snapshot is not available in this revision.")
  endif()
  add_subdirectory(revolverator)
endif()
