cmake_minimum_required(VERSION 3.20)
project(space_voyages LANGUAGES C CXX)
# Keep the exact corresponding baseline source bytes. GCC diagnoses its compact
# one-line statements differently from MSVC; retain that warning without making
# it fatal in the existing strict CPU fixtures. All other strict warnings stay.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
 add_compile_options(-Wno-error=misleading-indentation)
endif()
if(NOT OpenRACMod_DIR)
 set(OpenRACMod_DIR "${CMAKE_CURRENT_SOURCE_DIR}/support/sdk-1.14/cmake")
endif()
find_package(OpenRACMod REQUIRED)
include(CTest)
add_library(space_voyages_core STATIC src/catalog.cpp src/flight.cpp src/voyage.cpp src/asteroid_field.cpp src/asteroid_contacts.cpp src/ship_damage.cpp src/space_combat.cpp src/scene_audio.cpp src/garage_bridge.cpp src/campaign_progress.cpp)
set_target_properties(space_voyages_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(space_voyages_core PUBLIC src include)
target_compile_features(space_voyages_core PUBLIC cxx_std_17)
openrac_add_mod(space_voyages MANIFEST openrac-mod.json SOURCES src/mod.cpp src/assets.cpp src/renderer.cpp src/asteroid_renderer.cpp src/ship_effects.cpp src/projectile_renderer.cpp src/destination_client.cpp src/destination_gateway.cpp)
target_link_libraries(space_voyages PRIVATE space_voyages_core)
if(MSVC)
 target_compile_options(space_voyages PRIVATE /W4 /EHsc)
 target_compile_options(space_voyages_core PRIVATE /W4)
else()
 target_compile_options(space_voyages PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(UNIX AND NOT APPLE)
 file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/space_voyages.exports" "{ global: openrac_mod_query; openrac_mod_load; openrac_mod_unload; openrac_space_voyages_query; local: *; };\n")
 target_link_libraries(space_voyages PRIVATE dl)
endif()
if(BUILD_TESTING)
 foreach(test IN ITEMS flight voyage render_math input_adapter presentation disc_binding boarding_timeline asteroid_field asteroid_contacts asteroid_voyage diagnostic_pilot ship_damage health_meter space_combat reticle scene_audio garage_bridge campaign_progress destination_route ship_identity)
  add_executable(space_${test}_test tests/${test}_test.cpp)
  target_link_libraries(space_${test}_test PRIVATE space_voyages_core)
  add_test(NAME space_${test} COMMAND space_${test}_test)
 endforeach()
 add_executable(space_diagnostic_scene_control_test tests/diagnostic_scene_control_test.cpp)
 target_include_directories(space_diagnostic_scene_control_test PRIVATE src)
 target_compile_features(space_diagnostic_scene_control_test PRIVATE cxx_std_17)
 if(MSVC)
  target_compile_options(space_diagnostic_scene_control_test PRIVATE /W4 /WX /EHsc)
 else()
  target_compile_options(space_diagnostic_scene_control_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
 add_test(NAME space_diagnostic_scene_control COMMAND space_diagnostic_scene_control_test)
 # Exercise the production Session with a fake SDK, without starting a game
 # or GL context. This keeps handoff/reentry regressions in the normal suite.
 add_executable(space_session_test tests/session_test.cpp src/assets.cpp src/renderer.cpp src/asteroid_renderer.cpp src/ship_effects.cpp src/projectile_renderer.cpp src/destination_client.cpp src/destination_gateway.cpp)
 target_link_libraries(space_session_test PRIVATE space_voyages_core OpenRAC::mod)
 if(UNIX AND NOT APPLE)
  target_link_libraries(space_session_test PRIVATE dl)
 endif()
 if(MSVC)
  target_compile_definitions(space_session_test PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(space_session_test PRIVATE /W4 /WX /EHsc)
 endif()
 add_test(NAME space_session COMMAND space_session_test)
 add_executable(space_destination_client_test tests/destination_client_test.cpp src/destination_client.cpp src/destination_gateway.cpp)
 target_link_libraries(space_destination_client_test PRIVATE space_voyages_core OpenRAC::mod)
 add_test(NAME space_destination_client COMMAND space_destination_client_test)
 add_executable(space_destination_gateway_test tests/destination_gateway_test.cpp src/destination_gateway.cpp)
 target_link_libraries(space_destination_gateway_test PRIVATE space_voyages_core OpenRAC::mod)
 if(MSVC)
  target_compile_options(space_destination_gateway_test PRIVATE /W4 /WX /EHsc)
 endif()
 add_test(NAME space_destination_gateway COMMAND space_destination_gateway_test)
 if(WIN32)
  add_library(space_garage_provider_fixture SHARED tests/garage_provider_fixture.c)
  set_target_properties(space_garage_provider_fixture PROPERTIES OUTPUT_NAME veldin_garage_preview PREFIX "")
  add_executable(space_session_destination_test tests/session_destination_test.cpp src/assets.cpp src/renderer.cpp src/asteroid_renderer.cpp src/ship_effects.cpp src/projectile_renderer.cpp src/destination_client.cpp src/destination_gateway.cpp)
  target_link_libraries(space_session_destination_test PRIVATE space_voyages_core OpenRAC::mod)
  target_compile_definitions(space_session_destination_test PRIVATE _CRT_SECURE_NO_WARNINGS)
  if(MSVC)
   target_compile_options(space_session_destination_test PRIVATE /W4 /WX /EHsc)
  endif()
  add_dependencies(space_session_destination_test space_garage_provider_fixture)
  add_test(NAME space_session_destination COMMAND space_session_destination_test "$<TARGET_FILE:space_garage_provider_fixture>")
 endif()
 add_executable(space_session_audio_test tests/session_audio_test.cpp src/assets.cpp src/renderer.cpp src/asteroid_renderer.cpp src/ship_effects.cpp src/projectile_renderer.cpp src/destination_client.cpp src/destination_gateway.cpp)
 target_link_libraries(space_session_audio_test PRIVATE space_voyages_core OpenRAC::mod)
 if(UNIX AND NOT APPLE)
  target_link_libraries(space_session_audio_test PRIVATE dl)
 endif()
 if(MSVC)
  target_compile_definitions(space_session_audio_test PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(space_session_audio_test PRIVATE /W4 /WX /EHsc)
 endif()
 add_test(NAME space_session_audio COMMAND space_session_audio_test)
 add_executable(space_asteroid_renderer_test tests/asteroid_renderer_test.cpp src/assets.cpp src/asteroid_renderer.cpp)
 target_link_libraries(space_asteroid_renderer_test PRIVATE space_voyages_core)
 add_test(NAME space_asteroid_renderer COMMAND space_asteroid_renderer_test)
 add_executable(space_ship_effects_test tests/ship_effects_test.cpp src/ship_effects.cpp)
 target_link_libraries(space_ship_effects_test PRIVATE space_voyages_core)
 add_test(NAME space_ship_effects COMMAND space_ship_effects_test)
 add_executable(space_renderer_test tests/renderer_test.cpp src/renderer.cpp src/asteroid_renderer.cpp src/ship_effects.cpp src/projectile_renderer.cpp)
 target_link_libraries(space_renderer_test PRIVATE space_voyages_core)
 add_test(NAME space_renderer COMMAND space_renderer_test)
 add_executable(space_projectile_renderer_test tests/projectile_renderer_test.cpp src/projectile_renderer.cpp)
 target_link_libraries(space_projectile_renderer_test PRIVATE space_voyages_core)
 add_test(NAME space_projectile_renderer COMMAND space_projectile_renderer_test)
 if(MSVC)
  foreach(strict_target IN ITEMS space_ship_identity_test space_renderer_test space_ship_effects_test space_projectile_renderer_test space_scene_audio_test)
   target_compile_options(${strict_target} PRIVATE /W4 /WX /EHsc)
  endforeach()
 endif()
 add_executable(space_scene_audio_sdk_test tests/scene_audio_sdk_test.cpp)
 target_link_libraries(space_scene_audio_sdk_test PRIVATE space_voyages_core OpenRAC::mod)
 add_test(NAME space_scene_audio_sdk COMMAND space_scene_audio_sdk_test)
 add_executable(space_effect_assets_test tests/effect_assets_test.cpp src/assets.cpp)
 target_link_libraries(space_effect_assets_test PRIVATE space_voyages_core)
 add_test(NAME space_effect_assets COMMAND space_effect_assets_test)
 set_tests_properties(space_effect_assets PROPERTIES LABELS space_effect_assets)
 add_executable(space_audio_assets_test tests/audio_assets_test.cpp src/assets.cpp)
 target_link_libraries(space_audio_assets_test PRIVATE space_voyages_core)
 add_test(NAME space_audio_assets COMMAND space_audio_assets_test)
 set_tests_properties(space_audio_assets PROPERTIES LABELS space_audio_assets)
 set(SPACE_VOYAGES_ASSETS "" CACHE PATH "Locally generated original scene assets")
 if(EXISTS "${SPACE_VOYAGES_ASSETS}/assets.tsv")
  add_executable(space_assets_test tests/assets_test.cpp src/assets.cpp)
  target_link_libraries(space_assets_test PRIVATE space_voyages_core)
  add_test(NAME space_assets COMMAND space_assets_test "${SPACE_VOYAGES_ASSETS}")
  set(SPACE_VOYAGES_FULL_ASSETS "" CACHE PATH "Locally generated full galaxy assets")
  if(EXISTS "${SPACE_VOYAGES_FULL_ASSETS}/assets.tsv")
   add_test(NAME space_assets_full COMMAND space_assets_test "${SPACE_VOYAGES_FULL_ASSETS}")
  endif()
  set(SPACE_VOYAGES_BATCHED_ASSETS "" CACHE PATH "Locally generated batched galaxy assets")
  if(EXISTS "${SPACE_VOYAGES_BATCHED_ASSETS}/assets.tsv")
   add_test(NAME space_assets_batched COMMAND space_assets_test "${SPACE_VOYAGES_BATCHED_ASSETS}")
  endif()
  foreach(region IN ITEMS PAL NTSC)
   set(SPACE_VOYAGES_ASTEROID_${region}_ASSETS "" CACHE PATH "Locally prepared ${region} original asteroid scene assets")
   if(EXISTS "${SPACE_VOYAGES_ASTEROID_${region}_ASSETS}/assets.tsv")
    add_test(NAME space_assets_asteroid_${region} COMMAND space_assets_test "${SPACE_VOYAGES_ASTEROID_${region}_ASSETS}")
   endif()
   set(SPACE_VOYAGES_EFFECTS_${region}_ASSETS "" CACHE PATH "Locally prepared ${region} original ship-effect scene assets")
   if(EXISTS "${SPACE_VOYAGES_EFFECTS_${region}_ASSETS}/assets.tsv")
    add_test(NAME space_assets_effects_${region} COMMAND space_assets_test "${SPACE_VOYAGES_EFFECTS_${region}_ASSETS}")
   endif()
  endforeach()
 endif()
endif()
