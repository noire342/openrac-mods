cmake_minimum_required(VERSION 3.20)
project(arsenal_file_overlay LANGUAGES C CXX)
include(CTest)
add_library(file_overlay_core STATIC src/overlay_core.cpp)
target_compile_features(file_overlay_core PUBLIC cxx_std_17)
target_include_directories(file_overlay_core PUBLIC src include)
set_target_properties(file_overlay_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(file_overlay_core PRIVATE /W4 /EHsc)
else()
  target_compile_options(file_overlay_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  find_package(OpenRACMod 1.3 REQUIRED)
  if(NOT OPENRAC_MOD_PLATFORM STREQUAL "windows-x86_64")
    message(FATAL_ERROR "The file-overlay native POC supports Windows x64 only")
  endif()
  add_library(file_overlay_minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
    third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
  target_include_directories(file_overlay_minhook PUBLIC third_party/minhook/include)
  openrac_add_mod(arsenal_file_overlay MANIFEST openrac-mod.json SOURCES src/mod.cpp src/windows_overlay.cpp)
  target_link_libraries(arsenal_file_overlay PRIVATE file_overlay_core file_overlay_minhook bcrypt)
  target_compile_options(arsenal_file_overlay PRIVATE /W4 /EHsc)
  file(COPY third_party/minhook/LICENSE.txt DESTINATION "${CMAKE_BINARY_DIR}/package/arsenal_file_overlay/licenses/minhook")
  file(COPY third_party/minhook.lock.json docs/LICENSE-MIT.md DESTINATION "${CMAKE_BINARY_DIR}/package/arsenal_file_overlay/licenses")
endif()
if(BUILD_TESTING)
  add_executable(file_overlay_core_test tests/overlay_core_test.cpp)
  target_link_libraries(file_overlay_core_test PRIVATE file_overlay_core)
  add_test(NAME file_overlay_core COMMAND file_overlay_core_test)
endif()
