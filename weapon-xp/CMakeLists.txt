set(weapon_xp_core src/progression.cpp src/combat.cpp src/store.cpp src/ui.cpp src/events.cpp src/rules.cpp src/rules_file.cpp)
openrac_add_mod(weapon_xp MANIFEST openrac-mod.json
  SOURCES src/weapon_xp.cpp src/runtime.cpp src/hud_bridge.cpp src/native_hud.cpp src/native_upgrades.cpp src/api.cpp src/configuration_api.cpp ${weapon_xp_core})
target_include_directories(weapon_xp PRIVATE src include)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS openrac-mod.json)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/openrac-mod.json" weapon_xp_manifest)
string(JSON weapon_xp_version GET "${weapon_xp_manifest}" version)
target_compile_definitions(weapon_xp PRIVATE WEAPON_XP_MOD_VERSION="${weapon_xp_version}")
if(MSVC)
  target_compile_options(weapon_xp PRIVATE /W4 /EHsc)
else()
  target_compile_options(weapon_xp PRIVATE -Wall -Wextra -Wpedantic)
endif()
# The SDK's default ELF version script exposes only loader entry points. This mod also provides
# one explicitly versioned C service; keep every other implementation symbol hidden.
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/weapon_xp.exports"
    "{\n global: openrac_mod_query; openrac_mod_load; openrac_mod_unload; openrac_weapon_xp_query;\n local: *;\n};\n")
endif()
if(BUILD_TESTING)
  add_library(weapon_xp_rules_client_example OBJECT examples/rules_client.c)
  target_include_directories(weapon_xp_rules_client_example PRIVATE include)
  add_library(weapon_xp_core STATIC ${weapon_xp_core})
  target_include_directories(weapon_xp_core PUBLIC src include)
  target_compile_features(weapon_xp_core PUBLIC cxx_std_17)
  foreach(test core store ui events rules)
    add_executable(weapon_xp_${test}_test tests/${test}_test.cpp)
    target_link_libraries(weapon_xp_${test}_test PRIVATE weapon_xp_core)
    add_test(NAME weapon_xp_${test} COMMAND weapon_xp_${test}_test)
  endforeach()
  add_executable(weapon_xp_rewards_test tests/rewards_test.cpp)
  add_executable(weapon_xp_native_damage_test tests/native_damage_test.cpp)
  add_executable(weapon_xp_native_morph_model_test tests/native_morph_model_test.cpp)
  target_link_libraries(weapon_xp_native_morph_model_test PRIVATE weapon_xp_core OpenRAC::mod)
  add_test(NAME weapon_xp_native_morph_model COMMAND weapon_xp_native_morph_model_test)
  target_link_libraries(weapon_xp_native_damage_test PRIVATE weapon_xp_core OpenRAC::mod)
  add_test(NAME weapon_xp_native_damage COMMAND weapon_xp_native_damage_test)
  target_link_libraries(weapon_xp_rewards_test PRIVATE weapon_xp_core OpenRAC::mod)
  add_test(NAME weapon_xp_rewards COMMAND weapon_xp_rewards_test)
  add_executable(weapon_xp_bridge_test tests/bridge_test.cpp src/hud_bridge.cpp)
  target_link_libraries(weapon_xp_bridge_test PRIVATE weapon_xp_core OpenRAC::mod)
  add_test(NAME weapon_xp_bridge COMMAND weapon_xp_bridge_test)
  add_executable(weapon_xp_upgrades_test tests/upgrades_test.cpp src/native_upgrades.cpp)
  target_link_libraries(weapon_xp_upgrades_test PRIVATE weapon_xp_core OpenRAC::mod)
  add_test(NAME weapon_xp_upgrades COMMAND weapon_xp_upgrades_test)
  add_executable(weapon_xp_native_hud_test tests/native_hud_test.cpp src/native_hud.cpp)
  target_link_libraries(weapon_xp_native_hud_test PRIVATE weapon_xp_core OpenRAC::mod)
  add_test(NAME weapon_xp_native_hud COMMAND weapon_xp_native_hud_test)
  add_executable(weapon_xp_load_test tests/load_test.cpp)
  target_compile_definitions(weapon_xp_load_test PRIVATE
    WEAPON_XP_EXPECTED_VERSION="${weapon_xp_version}"
    WEAPON_XP_CONTRACT_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/installed-contracts.tsv")
  target_include_directories(weapon_xp_load_test PRIVATE src include)
  target_link_libraries(weapon_xp_load_test PRIVATE OpenRAC::mod weapon_xp_core ${CMAKE_DL_LIBS})
  add_dependencies(weapon_xp_load_test weapon_xp)
  if(WIN32)
    add_library(weapon_xp_observer SHARED examples/xp_observer.c)
    target_include_directories(weapon_xp_observer PRIVATE include)
    target_link_libraries(weapon_xp_observer PRIVATE OpenRAC::mod)
    add_dependencies(weapon_xp_load_test weapon_xp_observer)
    add_test(NAME weapon_xp_load COMMAND weapon_xp_load_test $<TARGET_FILE:weapon_xp> $<TARGET_FILE:weapon_xp_observer>)
  else()
    add_test(NAME weapon_xp_load COMMAND weapon_xp_load_test $<TARGET_FILE:weapon_xp>)
  endif()
endif()
