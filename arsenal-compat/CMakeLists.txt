cmake_minimum_required(VERSION 3.20)
project(arsenal_compat LANGUAGES C CXX)
include(CTest)
add_library(arsenal_compat_core STATIC src/provider.cpp src/hook_catalog.cpp)
target_compile_features(arsenal_compat_core PUBLIC cxx_std_17)
target_include_directories(arsenal_compat_core PUBLIC src include)
set_target_properties(arsenal_compat_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(arsenal_compat_core PRIVATE /W4 /EHsc)
else()
  target_compile_options(arsenal_compat_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  find_package(OpenRACMod 1.3 REQUIRED)
  if(NOT OPENRAC_MOD_PLATFORM STREQUAL "windows-x86_64")
    message(FATAL_ERROR "The native provider supports Windows x64 only; portable core tests need no native provider")
  endif()
  # Exact unmodified v1.3.4 files; own static instance, no shared DLL dependency.
  add_library(arsenal_minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c
    third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
  target_include_directories(arsenal_minhook PUBLIC third_party/minhook/include)
  openrac_add_mod(arsenal_compat MANIFEST openrac-mod.json SOURCES src/mod.cpp src/windows_backend.cpp)
  target_link_libraries(arsenal_compat PRIVATE arsenal_compat_core arsenal_minhook bcrypt)
  target_compile_options(arsenal_compat PRIVATE /W4 /EHsc)
  file(COPY third_party/minhook/LICENSE.txt DESTINATION "${CMAKE_BINARY_DIR}/package/arsenal_compat/licenses/minhook")
  file(COPY third_party/minhook.lock.json DESTINATION "${CMAKE_BINARY_DIR}/package/arsenal_compat/licenses")
  file(COPY docs/LICENSE-MIT.md DESTINATION "${CMAKE_BINARY_DIR}/package/arsenal_compat/licenses")
endif()
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(arsenal_compat_provider_test tests/provider_test.cpp)
  target_link_libraries(arsenal_compat_provider_test PRIVATE arsenal_compat_core Threads::Threads)
  add_test(NAME arsenal_compat_provider COMMAND arsenal_compat_provider_test)
endif()
