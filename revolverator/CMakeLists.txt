cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(revolverator LANGUAGES C CXX)
  find_package(OpenRACMod 1.3 REQUIRED)
  include(CTest)
endif()
add_library(revolverator_core STATIC src/controller.cpp src/arm_math.cpp src/attachment.cpp src/rotor_controller.cpp src/arsenal_state.cpp src/arsenal_store.cpp)
set_target_properties(revolverator_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(revolverator_core PUBLIC cxx_std_17)
target_include_directories(revolverator_core PUBLIC src include)
openrac_add_mod(revolverator MANIFEST openrac-mod.json SOURCES src/mod.cpp src/native_world.cpp src/native_arm.cpp src/native_ammunition.cpp src/native_vendor.cpp src/native_class_alias.cpp src/native_weapon_alias.cpp src/native_weapon_menu.cpp src/native_arsenal.cpp src/native_arsenal_save.cpp src/native_arsenal_xp.cpp src/native_runtime_hooks.cpp src/native_rotor.cpp src/hd_icon.cpp src/presentation.cpp)
target_link_libraries(revolverator PRIVATE revolverator_core ${CMAKE_DL_LIBS})
target_sources(revolverator PRIVATE src/native_asset_bundle.cpp src/native_compatibility_modules.cpp third_party/openrac/common/json.cpp)
target_include_directories(revolverator PRIVATE third_party/openrac)
target_compile_features(revolverator PRIVATE cxx_std_20)
if(WIN32)
  target_link_libraries(revolverator PRIVATE shell32)
endif()
if(UNIX AND NOT APPLE)
  # The SDK supplies a restrictive ELF version script. Include the optional
  # mod-to-mod service as well as the standard loader entry points.
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/revolverator.exports"
    "{ global: openrac_mod_query; openrac_mod_load; openrac_mod_unload; openrac_revolverator_query; local: *; };\n")
endif()
set(REVOLVERATOR_HD_ICON_DIR "" CACHE PATH "Locally generated native HD glyph data")
if(EXISTS "${REVOLVERATOR_HD_ICON_DIR}/revolverator_hd_icon_data.hpp")
  target_include_directories(revolverator PRIVATE "${REVOLVERATOR_HD_ICON_DIR}")
  target_compile_definitions(revolverator PRIVATE REVOLVERATOR_HD_ICON=1)
endif()
if(MSVC)
  target_compile_options(revolverator PRIVATE /W4 /EHsc)
  target_compile_options(revolverator_core PRIVATE /W4)
else()
  target_compile_options(revolverator PRIVATE -Wall -Wextra -Wpedantic)
endif()
