cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(revolverator LANGUAGES C CXX)
  find_package(OpenRACMod 1.3 REQUIRED)
  include(CTest)
endif()
add_library(revolverator_core STATIC src/controller.cpp src/arm_math.cpp src/attachment.cpp src/rotor_controller.cpp)
target_compile_features(revolverator_core PUBLIC cxx_std_17)
target_include_directories(revolverator_core PUBLIC src include)
openrac_add_mod(revolverator MANIFEST openrac-mod.json SOURCES src/mod.cpp src/native_world.cpp src/native_arm.cpp src/native_ammunition.cpp src/native_vendor.cpp src/native_class_alias.cpp src/native_rotor.cpp src/hd_icon.cpp src/presentation.cpp)
target_link_libraries(revolverator PRIVATE revolverator_core)
if(UNIX AND NOT APPLE)
  # The SDK supplies a restrictive ELF version script. Include the optional
  # mod-to-mod service as well as the standard loader entry points.
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/revolverator.exports"
    "{ global: openrac_mod_query; openrac_mod_load; openrac_mod_unload; openrac_revolverator_query; local: *; };\n")
endif()
set(REVOLVERATOR_HD_ICON_DIR "" CACHE PATH "Locally generated native HD glyph data")
if(EXISTS "${REVOLVERATOR_HD_ICON_DIR}/revolverator_hd_icon_data.hpp")
  target_include_directories(revolverator PRIVATE "${REVOLVERATOR_HD_ICON_DIR}")
  target_compile_definitions(revolverator PRIVATE REVOLVERATOR_HD_ICON=1)
endif()
if(MSVC)
  target_compile_options(revolverator PRIVATE /W4 /EHsc)
  target_compile_options(revolverator_core PRIVATE /W4)
else()
  target_compile_options(revolverator PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  add_library(revolverator_client_example OBJECT examples/state_client.c)
  target_include_directories(revolverator_client_example PRIVATE include)
  add_executable(revolverator_controller_test tests/controller_test.cpp)
  target_link_libraries(revolverator_controller_test PRIVATE revolverator_core)
  add_test(NAME revolverator_controller COMMAND revolverator_controller_test)
  add_executable(revolverator_pose_test tests/pose_test.cpp)
  target_link_libraries(revolverator_pose_test PRIVATE revolverator_core)
  add_test(NAME revolverator_pose COMMAND revolverator_pose_test)
  add_executable(revolverator_ammunition_test tests/native_ammunition_test.cpp src/native_ammunition.cpp)
  target_compile_features(revolverator_ammunition_test PRIVATE cxx_std_17)
  target_include_directories(revolverator_ammunition_test PRIVATE src)
  target_link_libraries(revolverator_ammunition_test PRIVATE OpenRAC::mod)
  add_test(NAME revolverator_ammunition COMMAND revolverator_ammunition_test)
  add_executable(revolverator_vendor_test tests/native_vendor_test.cpp src/native_vendor.cpp)
  target_compile_features(revolverator_vendor_test PRIVATE cxx_std_17)
  target_include_directories(revolverator_vendor_test PRIVATE src)
  target_link_libraries(revolverator_vendor_test PRIVATE OpenRAC::mod)
  add_test(NAME revolverator_vendor COMMAND revolverator_vendor_test)
  add_executable(revolverator_rotor_test tests/rotor_controller_test.cpp)
  target_link_libraries(revolverator_rotor_test PRIVATE revolverator_core)
  add_test(NAME revolverator_rotor COMMAND revolverator_rotor_test)
  add_executable(revolverator_arena_test tests/native_arena_test.cpp)
  target_link_libraries(revolverator_arena_test PRIVATE revolverator_core)
  add_test(NAME revolverator_arena COMMAND revolverator_arena_test)
  add_executable(revolverator_localization_test tests/localization_test.cpp)
  target_link_libraries(revolverator_localization_test PRIVATE revolverator_core)
  add_test(NAME revolverator_localization COMMAND revolverator_localization_test)
  foreach(adapter IN ITEMS native_rotor native_class_alias hd_icon)
    add_executable(revolverator_${adapter}_test tests/${adapter}_test.cpp src/${adapter}.cpp)
    target_link_libraries(revolverator_${adapter}_test PRIVATE revolverator_core OpenRAC::mod)
    add_test(NAME revolverator_${adapter} COMMAND revolverator_${adapter}_test)
  endforeach()
endif()
