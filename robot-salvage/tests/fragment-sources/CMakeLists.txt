cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(RobotSalvageFragmentSourceTests LANGUAGES C CXX)
 find_package(OpenRACMod 1.3 REQUIRED)
 include(CTest)
endif()
add_executable(robot_salvage_fragment_sources_test fragment_sources_test.cpp ../../src/native_fragment_sources.cpp ../../src/core.cpp ../../src/builtin_catalog.cpp)
target_include_directories(robot_salvage_fragment_sources_test PRIVATE ../../src ../../include)
target_link_libraries(robot_salvage_fragment_sources_test PRIVATE OpenRAC::mod)
target_compile_features(robot_salvage_fragment_sources_test PRIVATE cxx_std_17)
if(MSVC)
 target_compile_options(robot_salvage_fragment_sources_test PRIVATE /W4 /EHsc)
else()
 target_compile_options(robot_salvage_fragment_sources_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME robot_salvage_fragment_sources COMMAND robot_salvage_fragment_sources_test)
