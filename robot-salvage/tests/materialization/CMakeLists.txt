cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(RobotSalvageMaterializationTests LANGUAGES CXX)
 include(CTest)
endif()
add_executable(robot_salvage_materialization_test materialization_test.cpp ../../src/materialization_policy.cpp ../../src/core.cpp)
target_include_directories(robot_salvage_materialization_test PRIVATE ../../src)
target_compile_features(robot_salvage_materialization_test PRIVATE cxx_std_17)
if(MSVC)
 target_compile_options(robot_salvage_materialization_test PRIVATE /W4 /EHsc)
else()
 target_compile_options(robot_salvage_materialization_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME robot_salvage_materialization COMMAND robot_salvage_materialization_test)
