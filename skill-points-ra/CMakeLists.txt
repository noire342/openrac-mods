cmake_minimum_required(VERSION 3.20)
project(skill_points_ra LANGUAGES C CXX)
if(NOT OpenRACMod_DIR)
  if(OPENRAC_SDK)
    set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
  else()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk.lock" sdk_lock)
    string(JSON sdk_version GET "${sdk_lock}" version)
    set(OpenRACMod_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/openrac-mod-sdk-${sdk_version}/cmake")
  endif()
endif()
find_package(OpenRACMod 1.3 REQUIRED)
option(SP_BUILD_QA_PROBE "Build isolated native QA probe (never shipped)" OFF)
if(SP_BUILD_QA_PROBE)
  openrac_add_mod(skill_points_qa MANIFEST tests/qa-mod.json SOURCES tests/skill_points_probe.cpp)
endif()
include(CTest)
file(GLOB ra_runtime CONFIGURE_DEPENDS vendor/rcheevos/src/runtime/*.c)
add_library(ra_runtime STATIC ${ra_runtime}
  vendor/rcheevos/src/util/md5.c vendor/rcheevos/src/util/rc_compat.c
  vendor/rcheevos/src/util/rc_util.c vendor/rcheevos/src/util/rc_version.c)
target_include_directories(ra_runtime PUBLIC vendor/rcheevos/include)
set_target_properties(ra_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
target_compile_definitions(ra_runtime PRIVATE RC_STATIC)
add_library(skill_points_core STATIC src/catalogue_data.cpp src/ledger.cpp src/engine.cpp src/store.cpp src/menu.cpp)
target_include_directories(skill_points_core PUBLIC src include)
target_compile_features(skill_points_core PUBLIC cxx_std_17)
target_link_libraries(skill_points_core PUBLIC ra_runtime)
set_target_properties(skill_points_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
openrac_add_mod(skill_points_ra MANIFEST openrac-mod.json SOURCES src/plugin.cpp src/runtime.cpp src/service.cpp src/native_ui.cpp src/pause_access.cpp)
target_include_directories(skill_points_ra PRIVATE src include)
target_link_libraries(skill_points_ra PRIVATE skill_points_core)
if(MSVC)
  target_compile_options(skill_points_core PRIVATE /W4 /EHsc)
  target_compile_options(skill_points_ra PRIVATE /W4 /EHsc)
else()
  target_compile_options(skill_points_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(skill_points_ra PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/skill_points_ra.exports"
    "{ global: openrac_mod_query; openrac_mod_load; openrac_mod_unload; openrac_skill_points_query; local: *; };\n")
endif()
if(BUILD_TESTING)
  foreach(test engine ledger menu)
    add_executable(skill_points_${test}_test tests/${test}_test.cpp)
    target_link_libraries(skill_points_${test}_test PRIVATE skill_points_core)
    add_test(NAME skill_points_${test} COMMAND skill_points_${test}_test)
  endforeach()
  add_executable(skill_points_native_ui_test tests/native_ui_test.cpp src/native_ui.cpp src/pause_access.cpp)
  target_link_libraries(skill_points_native_ui_test PRIVATE skill_points_core OpenRAC::mod)
  add_test(NAME skill_points_native_ui COMMAND skill_points_native_ui_test)
  add_executable(skill_points_load_test tests/load_test.cpp)
  target_link_libraries(skill_points_load_test PRIVATE skill_points_core OpenRAC::mod ${CMAKE_DL_LIBS})
  add_dependencies(skill_points_load_test skill_points_ra)
  if(WIN32)
    add_library(skill_points_observer SHARED examples/achievement_observer.c)
    target_include_directories(skill_points_observer PRIVATE include)
    target_link_libraries(skill_points_observer PRIVATE OpenRAC::mod)
    add_dependencies(skill_points_load_test skill_points_observer)
    add_test(NAME skill_points_load COMMAND skill_points_load_test $<TARGET_FILE:skill_points_ra> $<TARGET_FILE:skill_points_observer>)
  else()
    add_test(NAME skill_points_load COMMAND skill_points_load_test $<TARGET_FILE:skill_points_ra>)
  endif()
endif()
