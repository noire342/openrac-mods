cmake_minimum_required(VERSION 3.20)
project(cinematic_journeys LANGUAGES CXX)
if(NOT OpenRACMod_DIR)
  if(NOT OPENRAC_SDK)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk.lock" sdk_lock)
    string(JSON sdk_version GET "${sdk_lock}" version)
    set(OPENRAC_SDK "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/openrac-mod-sdk-${sdk_version}")
  endif()
  set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
endif()
find_package(OpenRACMod 1.3 REQUIRED)
include(CTest)
option(CINEMATIC_BUILD_QA "Build native QA fixture (never distributed)" OFF)
if(CINEMATIC_BUILD_QA)
  openrac_add_mod(cinematic_qa MANIFEST tests/qa-mod.json SOURCES tests/native_probe.cpp)
endif()
add_library(cinematic_policy STATIC src/policy.cpp)
target_include_directories(cinematic_policy PUBLIC src)
target_compile_features(cinematic_policy PUBLIC cxx_std_17)
set_target_properties(cinematic_policy PROPERTIES POSITION_INDEPENDENT_CODE ON)
openrac_add_mod(cinematic_journeys MANIFEST openrac-mod.json
  SOURCES src/plugin.cpp src/runtime.cpp)
target_link_libraries(cinematic_journeys PRIVATE cinematic_policy)
if(MSVC)
  target_compile_options(cinematic_policy PRIVATE /W4 /WX /EHsc)
  target_compile_options(cinematic_journeys PRIVATE /W4 /WX /EHsc)
else()
  target_compile_options(cinematic_policy PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(cinematic_journeys PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
  add_executable(cinematic_policy_test tests/policy_test.cpp)
  target_link_libraries(cinematic_policy_test PRIVATE cinematic_policy)
  add_test(NAME cinematic_policy COMMAND cinematic_policy_test)
  add_executable(cinematic_runtime_test tests/runtime_test.cpp src/runtime.cpp)
  target_link_libraries(cinematic_runtime_test PRIVATE cinematic_policy OpenRAC::mod)
  add_test(NAME cinematic_runtime COMMAND cinematic_runtime_test)
  add_executable(cinematic_lifecycle_test tests/lifecycle_test.cpp src/plugin.cpp src/runtime.cpp)
  target_link_libraries(cinematic_lifecycle_test PRIVATE cinematic_policy OpenRAC::mod)
  add_test(NAME cinematic_lifecycle COMMAND cinematic_lifecycle_test)
endif()
